# ======================================================================
# FLAME Detection Rule: Return Without Receipt Clustering by Customer or Region
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: medium
# CFPF Phase: P5
# Threat Paths: TP-0031
#
# Description:
#   Detects clusters of product returns submitted without receipts or order
#   references from the same customer identity or geographic region.
#   Organized return fraud rings often exploit no-receipt return policies
#   by using stolen merchandise or purchasing items specifically for
#   return, concentrating activity across nearby store locations or using
#   multiple accounts linked to the same region.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: ecommerce
#   service: returns_processing
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection:
#     event_type: return_processed
#     receipt_provided: false
#     order_reference_matched: false
#   aggregation_customer:
#     condition: selection | count() by customer_account_id > 4
#   aggregation_region:
#     condition: selection | count() by store_region_code > 15
#   condition: selection | aggregation_customer or selection | aggregation_region
#   timeframe: 30d
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Customers who frequently lose receipts for legitimate purchases
#   - Holiday season returns where gift recipients do not have original receipts
#
# ======================================================================
# Implementation Notes:
#   - Review the condition syntax and implement using your SIEM's
#     native aggregation/correlation capabilities
#   - Splunk: Use 'stats' or 'transaction' commands
#   - Elastic: Use EQL sequences or aggregation queries
#   - Sentinel: Use KQL summarize/make-series operators
# ======================================================================
