# ======================================================================
# FLAME Detection Rule: Cloud Phone Behavioral Anomaly — Device-Environment Decorrelation Detection
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: high
# CFPF Phase: P2
# Threat Paths: TP-0059, TP-0022
#
# Description:
#   Detects cloud phone behavioral anomalies indicative of industrial-scale
#   dropper/mule account creation. Cloud phones — remote-access Android
#   devices running genuine firmware on ARM hardware in data centers — pass
#   hardware attestation and present authentic device signals, making them
#   invisible to traditional emulator detection. This rule uses a composite
#   scoring approach combining device-environment decorrelation signals:
#   perpetual full battery, absence of accelerometer motion during active
#   sessions, presence of VPN/proxy and vendor management apps, unusually
#   high banking app density, and recent device first-seen dates. Group-IB
#   reports cloud phone rental from $0.10/hour with pre-verified dropper
#   accounts selling for $50–200 on darknet markets. UK APP fraud losses
#   reached £485.2M in 2023 with dropper accounts as the top contributing
#   factor.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: banking
#   service: mobile_device_telemetry
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection:
#     device_platform: android
#   filter_battery:
#     battery_level: 100
#   filter_motion:
#     accelerometer_motion_events: 0
#   filter_vpn:
#     vpn_proxy_app_installed: true
#   filter_banking_density:
#     banking_app_count|gte: 3
#   filter_vendor_app:
#     vendor_management_app_present:
#     - LDC Store
#     - LD Assistant
#   filter_new_device:
#     device_first_seen_days|lt: 30
#   timeframe: session
#   condition: 'selection AND filter_battery AND filter_motion during active session AND
#     (filter_vpn OR filter_vendor_app) AND filter_banking_density AND filter_new_device
#   
#     '
#   composite_score: 'Assign weighted score: battery_level=100 (+20), accelerometer_motion_events=0
#     during active session (+25), vpn_proxy_app_installed (+15), banking_app_count >=
#     3 (+15), vendor_management_app_present (+20), device_first_seen < 30 days (+5).
#     Alert threshold: score >= 60.
#   
#     '
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Devices in static charging cradles
#   - Accessibility-configured tablets used for banking
#
