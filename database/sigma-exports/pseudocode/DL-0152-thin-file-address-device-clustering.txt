# ======================================================================
# FLAME Detection Rule: Thin-File Identity Clustering — Shared Address, Phone, or Device Across New Applicants
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: high
# CFPF Phase: P2
# Threat Paths: TP-0003, TP-0029
#
# Description:
#   Detects clusters of thin-file credit applicants sharing physical
#   addresses, phone numbers, or device fingerprints — a hallmark indicator
#   of synthetic identity ring activity. Fraud rings fabricate multiple
#   synthetic identities that share infrastructure (mailing addresses,
#   burner phones, devices used for applications). When 3+ new applicants
#   with thin credit files (< 12 months history, < 3 tradelines) share an
#   address, phone, or device within a 180-day window, this indicates
#   coordinated identity fabrication rather than coincidental overlap. The
#   Federal Reserve identifies address and device clustering as primary
#   synthetic identity detection signals.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: banking
#   service: account_onboarding
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection_thin_file:
#     credit_history_months|lt: 12
#     tradeline_count|lt: 3
#   condition: 'count(distinct applicant_id) by shared_attribute >= 3 where shared_attribute
#     in (address, phone, device_fingerprint) within 180 days AND selection_thin_file
#   
#     '
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Shared housing (college dormitories, military barracks, assisted living facilities)
#   - Family members with thin credit files applying from the same household
#   - New immigrants sharing temporary housing while establishing credit
#
