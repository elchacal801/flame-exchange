# ======================================================================
# FLAME Detection Rule: SIM Swap Indicator Chain
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: critical
# CFPF Phase: P3
# Threat Paths: TP-0008
#
# Description:
#   Detects the full SIM swap attack chain: phone number change followed by
#   password reset followed by high-value account action in rapid sequence.
#   This multi-stage correlation captures the complete attack lifecycle
#   where the attacker first takes over the phone number, then uses it to
#   reset the account password via SMS OTP, and finally executes a
#   high-value transfer or account modification.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: banking
#   service: identity_management
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   stage1_sim_change:
#     event_type|contains:
#     - phone_number_change
#     - sim_swap
#     - mobile_update
#   stage2_password_reset:
#     event_type: password_reset
#     reset_method: sms_otp
#   stage3_high_value_action:
#     event_type|contains:
#     - wire_transfer
#     - beneficiary_add
#     - address_change
#     - email_change
#     transaction_amount|gte: 1000
#   timeframe: 24h
#   condition: 'stage1_sim_change | followed_by stage2_password_reset | followed_by stage3_high_value_action
#     on user_id within 24h
#   
#     '
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Legitimate users who change carriers, reset a forgotten password, and then make a large purchase in the same day
#
# ======================================================================

#   - Review the condition syntax and implement using your SIEM's
#     native aggregation/correlation capabilities
#   - Splunk: Use 'stats' or 'transaction' commands
#   - Elastic: Use EQL sequences or aggregation queries
#   - Sentinel: Use KQL summarize/make-series operators
