# ======================================================================
# FLAME Detection Rule: Disability Insurance Fraud — Fabricated Medical Documentation Provider Anomaly
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: high
# CFPF Phase: P4
# Threat Paths: TP-0010
#
# Description:
#   Detects patterns indicative of fabricated medical documentation
#   supporting fraudulent disability claims by identifying two key signals:
#   (1) a single medical provider billing across multiple claimants with
#   identical or near-identical diagnosis codes (ICD-10), suggesting
#   template-driven fabricated documentation rather than genuine clinical
#   variation, and (2) claims supported by providers located outside the
#   claimant's geographic area, indicating the claimant traveled
#   specifically to a complicit provider or the documentation was
#   fabricated remotely. DOJ prosecutions reveal organized disability fraud
#   rings where complicit providers generate documentation for dozens of
#   fictitious or exaggerated claims, with cumulative payouts in the
#   millions. The long-tail nature of disability payments means a single
#   fraudulent claim can generate $100,000+ in benefits before detection.
#   Generative AI tools now enable fabrication of convincing medical
#   documentation without provider collusion, representing an emerging
#   escalation vector.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: insurance
#   service: claims_management
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection:
#     claim_type:
#     - short_term_disability
#     - long_term_disability
#     claim_status:
#     - open
#     - approved
#     - in_payment
#   filter_provider_diagnosis_concentration:
#     provider_identical_diagnosis_claimant_count|gte: 3
#   filter_provider_geographic_mismatch:
#     provider_claimant_distance_miles|gte: 150
#   timeframe: 365d
#   condition: 'selection AND (filter_provider_diagnosis_concentration OR filter_provider_geographic_mismatch)
#   
#     '
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Specialist providers legitimately treating patients with the same condition (e.g., orthopedic surgeons with high back injury volume)
#   - Rural claimants who must travel long distances to see specialists not available locally
#   - Telehealth providers with geographically dispersed patient bases
#   - Occupational health providers serving a single large employer with common workplace injuries
