# ======================================================================
# FLAME Detection Rule: Split Transfer Chain Detection
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: high
# CFPF Phase: P4
# Threat Paths: TP-0002, TP-0006, TP-0020
#
# Description:
#   Detects when a large inbound amount is split into multiple smaller
#   outbound transfers to different beneficiaries in sequence. This
#   layering technique is used to obscure the origin and destination of
#   fraudulently obtained funds.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: banking
#   service: wire_system
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection_large_inbound:
#     event_type:
#     - WIRE_CREDIT
#     - ACH_CREDIT
#     transaction_amount|gte: 25000
#   selection_multiple_outbound:
#     event_type:
#     - WIRE_INITIATION
#     - ACH_DEBIT
#     - INTERNAL_TRANSFER
#     transaction_amount|lt: 10000
#   timeframe: 48h
#   condition: selection_large_inbound | near selection_multiple_outbound by account_id
#     | count(distinct beneficiary_id) by account_id > 2
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Businesses receiving large payments and distributing to multiple sub-contractors
#   - Investment accounts rebalancing across multiple fund positions
#
# ======================================================================
# Implementation Notes:
#   - Review the condition syntax and implement using your SIEM's
#     native aggregation/correlation capabilities
#   - Splunk: Use 'stats' or 'transaction' commands
#   - Elastic: Use EQL sequences or aggregation queries
#   - Sentinel: Use KQL summarize/make-series operators
# ======================================================================
