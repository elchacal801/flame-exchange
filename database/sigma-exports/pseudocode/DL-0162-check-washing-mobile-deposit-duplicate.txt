# ======================================================================
# FLAME Detection Rule: Check Washing — Altered Check Mobile Deposit with Duplicate or MICR Inconsistency
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: high
# CFPF Phase: P4
# Threat Paths: TP-0009
#
# Description:
#   Detects mobile deposits of potentially washed or altered checks by
#   identifying two key indicators: (1) the same check number deposited at
#   multiple financial institutions within a short window, indicating
#   duplicate deposit fraud, and (2) check amounts inconsistent with the
#   MICR (Magnetic Ink Character Recognition) line data, suggesting the
#   payee and amount fields were chemically washed and rewritten while the
#   MICR line was preserved. FinCEN data shows check fraud SARs surged
#   200%+ from 2021-2023, driven by USPS mail theft and the ease of mobile
#   deposit. Washed checks deposited via mobile apps exploit the lag
#   between deposit and check clearing under Reg CC funds availability
#   requirements. The Recorded Future 2025 report documented a geographic
#   shift in stolen check activity toward the Midwest and suburban/rural
#   USPS routes.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: banking
#   service: check_processing
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection:
#     deposit_channel:
#     - mobile
#     - remote_deposit_capture
#   filter_duplicate_check:
#     check_number_deposit_count|gte: 2
#   filter_micr_mismatch:
#     micr_amount_match: false
#   filter_account_age:
#     account_age_days|lt: 30
#   timeframe: 7d
#   condition: 'selection AND (filter_duplicate_check OR filter_micr_mismatch) AND filter_account_age
#   
#     '
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Legitimate re-deposit of returned checks (e.g., NSF item re-presented by payee)
#   - Minor MICR encoding errors from check printing vendors on legitimate checks
#   - Business customers with high-volume remote deposit capture processing
#
