# ======================================================================
# FLAME Detection Rule: MFA Push Bombing
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: high
# CFPF Phase: P2
# Threat Paths: TP-0008, TP-0001
#
# Description:
#   Detects rapid repeated MFA push notifications sent to the same user, a
#   technique known as MFA fatigue or push bombing. Attackers who have
#   obtained valid credentials bombard the victim with push authentication
#   requests, hoping the user will approve one out of frustration or
#   confusion, thereby granting the attacker access.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: banking
#   service: mfa_logs
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection:
#     event_type: mfa_push_sent
#     mfa_method: push
#   selection_denied:
#     mfa_result|contains:
#     - denied
#     - timeout
#     - ignored
#   timeframe: 10m
#   condition: 'selection | count() by user_id > 5 and count(selection_denied) by user_id
#     > 3
#   
#     '
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Connectivity issues causing MFA push retries from legitimate authentication flows
#   - Users who accidentally trigger multiple login attempts from different app instances
#
# ======================================================================
# Implementation Notes:
#   - Review the condition syntax and implement using your SIEM's
#     native aggregation/correlation capabilities
#   - Splunk: Use 'stats' or 'transaction' commands
#   - Elastic: Use EQL sequences or aggregation queries
#   - Sentinel: Use KQL summarize/make-series operators
# ======================================================================
