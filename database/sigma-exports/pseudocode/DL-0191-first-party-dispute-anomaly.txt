# ======================================================================
# FLAME Detection Rule: First-Party Dispute Anomaly — Chargeback with Delivery Confirmation, Device Match, and No Merchant Contact
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: high
# CFPF Phase: P4
# Threat Paths: TP-0075
#
# Description:
#   Detects high-confidence friendly fraud by identifying chargebacks where
#   three independent indicators contradict the dispute claim: (1) delivery
#   was confirmed for the original order, (2) the device fingerprint filing
#   the dispute matches the device used for the original purchase, and (3)
#   no merchant contact was attempted before the chargeback was filed. The
#   triple correlation of delivery evidence, device match, and merchant
#   bypass is a strong signal that the cardholder is the original purchaser
#   who received the goods and chose to file a chargeback rather than a
#   legitimate return. 84% of consumers find chargebacks simpler than
#   contacting merchants, creating systemic abuse of the dispute mechanism.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: payments
#   service: dispute_management
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection:
#     dispute_type:
#     - chargeback
#   filter_delivery:
#     delivery_confirmed: true
#   filter_device_match:
#     device_fingerprint_match: true
#   filter_no_merchant_contact:
#     merchant_contact_prior: false
#   timeframe: 30d
#   condition: 'selection AND filter_delivery AND filter_device_match AND filter_no_merchant_contact
#   
#     '
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Legitimate disputes where the product was defective but cardholder did not contact merchant
#   - Shared devices where a different household member made the purchase
#   - Delivery marked as confirmed but actually misdelivered (e.g., left at wrong address)
#
