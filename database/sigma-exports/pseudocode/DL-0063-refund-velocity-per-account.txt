# ======================================================================
# FLAME Detection Rule: Abnormally High Refund Rate Per Customer Account
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: medium
# CFPF Phase: P5
# Threat Paths: TP-0031, TP-0013
#
# Description:
#   Detects customer accounts with an abnormally high refund-to-order ratio
#   over a rolling window. Accounts exceeding a refund rate of 40% across
#   10 or more orders within 90 days are flagged as potential refund
#   abusers or participants in refunding-as-a-service (RaaS) schemes, where
#   professional refunders exploit return policies systematically.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: ecommerce
#   service: order_management
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection:
#     event_type|contains:
#     - order_completed
#     - refund_issued
#   aggregation:
#     condition: 'selection
#   
#       | groupby(customer_account_id)
#   
#       | where count(event_type=''refund_issued'') / count(event_type=''order_completed'')
#       > 0.4
#   
#       | where count(event_type=''order_completed'') >= 10
#   
#       '
#   timeframe: 90d
#   condition: selection | aggregation
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Customers with legitimate sizing issues making frequent clothing returns
#   - Product quality issues causing high return rates for specific SKUs across multiple customers
#
# ======================================================================
# Implementation Notes:
#   - Review the condition syntax and implement using your SIEM's
#     native aggregation/correlation capabilities
#   - Splunk: Use 'stats' or 'transaction' commands
#   - Elastic: Use EQL sequences or aggregation queries
#   - Sentinel: Use KQL summarize/make-series operators
# ======================================================================
