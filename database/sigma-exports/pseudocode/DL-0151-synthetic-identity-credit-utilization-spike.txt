# ======================================================================
# FLAME Detection Rule: Synthetic Identity Credit Utilization Spike — Low-and-Slow Buildup Followed by Bust-Out
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: high
# CFPF Phase: P4
# Threat Paths: TP-0003, TP-0016
#
# Description:
#   Detects the characteristic synthetic identity bust-out pattern:
#   prolonged low credit utilization (< 35% for 6+ months) followed by a
#   rapid spike to near-maximum utilization (> 80%) within 30 days.
#   Synthetic identity fraudsters nurture credit profiles over 12-24 months
#   with steady payments and low utilization, then execute a coordinated
#   bust-out where all available credit is maxed simultaneously. The
#   Federal Reserve estimates synthetic identity fraud costs U.S. lenders
#   $6B+ annually. The discontinuous utilization jump from a stable
#   baseline is the hallmark behavioral indicator.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: banking
#   service: credit_monitoring
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection:
#     account_type:
#     - consumer_credit
#     - credit_card
#   filter_utilization_history:
#     trailing_6m_avg_utilization_pct|lt: 35
#   filter_current_spike:
#     current_utilization_pct|gte: 80
#   filter_previous_month:
#     previous_month_utilization_pct|lt: 50
#   timeframe: 30d
#   condition: 'selection AND filter_utilization_history AND filter_current_spike AND
#     filter_previous_month
#   
#     '
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Legitimate large purchases (wedding, home renovation) causing temporary utilization spike
#   - Consumers consolidating debt via balance transfer to a single card
#   - Emergency medical expenses
#
