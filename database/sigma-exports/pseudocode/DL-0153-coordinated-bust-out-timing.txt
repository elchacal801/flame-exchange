# ======================================================================
# FLAME Detection Rule: Coordinated Bust-Out Timing — Simultaneous Max-Utilization Across Multiple Products
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: critical
# CFPF Phase: P4
# Threat Paths: TP-0003
#
# Description:
#   Detects coordinated bust-out execution where the same identity (or
#   linked identities) max out credit across multiple products within a
#   72-hour window. Synthetic identity rings time their bust-out to extract
#   maximum value before accounts are frozen: cash advances, balance
#   transfers to controlled accounts, and high-value purchases for resale
#   all occur within a narrow window. When 3+ credit products associated
#   with the same identity show utilization above 90% within 72 hours, and
#   at least one involves a cash advance or balance transfer, this is a
#   high-confidence bust-out indicator.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: banking
#   service: transaction_monitoring
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection_high_utilization:
#     current_utilization_pct|gte: 90
#   selection_cash_advance:
#     transaction_type:
#     - CASH_ADVANCE
#     - BALANCE_TRANSFER
#     - CONVENIENCE_CHECK
#   timeframe: 72h
#   condition: 'count(distinct account_id) by identity_id >= 3 AND selection_high_utilization
#     AND at least one selection_cash_advance within timeframe
#   
#     '
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Consumers aggressively using multiple cards for large planned purchases (rare)
#   - Business owners using multiple personal cards for business expenses during cash flow crunch
#

#   - Elastic: Use EQL sequences or aggregation queries
#   - Sentinel: Use KQL summarize/make-series operators
