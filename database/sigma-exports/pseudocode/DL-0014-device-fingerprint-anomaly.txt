# ======================================================================
# FLAME Detection Rule: Device Fingerprint Anomaly
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: medium
# CFPF Phase: P2
# Threat Paths: TP-0001, TP-0008
#
# Description:
#   Detects a known device fingerprint presenting with changed underlying
#   characteristics such as OS version, screen resolution, timezone, or
#   installed fonts. Attackers using antidetect browsers (Multilogin,
#   GoLogin, Dolphin Anty) import stolen device fingerprints but often have
#   mismatches in lower-level hardware attributes that this rule
#   correlates.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: banking
#   service: web_gateway
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection:
#     event_type: session_start
#     device_fingerprint|exists: true
#   condition: "selection | where device_fingerprint in known_device_fingerprints and\
#     \ (\n  os_version != baseline.os_version\n  or screen_resolution != baseline.screen_resolution\n\
#     \  or timezone_offset != baseline.timezone_offset\n  or webgl_renderer != baseline.webgl_renderer\n\
#     \  or canvas_hash != baseline.canvas_hash\n)\n"
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Users who updated their OS or browser, changing underlying fingerprint attributes
#   - Legitimate device reconfigurations such as connecting external monitors
#
# ======================================================================
# Implementation Notes:
#   - Review the condition syntax and implement using your SIEM's
#     native aggregation/correlation capabilities
#   - Splunk: Use 'stats' or 'transaction' commands
#   - Elastic: Use EQL sequences or aggregation queries
#   - Sentinel: Use KQL summarize/make-series operators
# ======================================================================
