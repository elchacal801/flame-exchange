# ======================================================================
# FLAME Detection Rule: Transaction Amount Clustering Near Reporting Threshold
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: high
# CFPF Phase: P5
# Threat Paths: TP-0002, TP-0020
#
# Description:
#   Detects transactions that cluster just below regulatory reporting
#   thresholds ($3,000 for international transfers, $10,000 for CTR
#   filing). Structuring transactions to evade reporting requirements is
#   itself a federal offense and a strong indicator of money laundering or
#   fraud monetization.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: banking
#   service: transaction_monitoring
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection_under_10k:
#     event_type:
#     - WIRE_INITIATION
#     - ACH_DEBIT
#     - CASH_DEPOSIT
#     transaction_amount|gte: 9000
#     transaction_amount|lt: 10000
#   selection_under_3k:
#     event_type:
#     - WIRE_INITIATION
#     - INTERNATIONAL_TRANSFER
#     transaction_amount|gte: 2800
#     transaction_amount|lt: 3000
#   timeframe: 7d
#   condition: (selection_under_10k
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Small business owners with legitimately recurring transactions that happen to fall near thresholds
#   - Payroll deposits that naturally cluster at consistent amounts
#
# ======================================================================
# Implementation Notes:
#   - Review the condition syntax and implement using your SIEM's
#     native aggregation/correlation capabilities
#   - Splunk: Use 'stats' or 'transaction' commands
#   - Elastic: Use EQL sequences or aggregation queries
#   - Sentinel: Use KQL summarize/make-series operators
# ======================================================================
