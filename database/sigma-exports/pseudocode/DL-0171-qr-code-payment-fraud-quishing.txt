# ======================================================================
# FLAME Detection Rule: QR Code Payment Fraud (Quishing) — Malicious QR Destination to Credential Harvesting Infrastructure
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: medium
# CFPF Phase: P2
# Threat Paths: TP-0051
#
# Description:
#   Detects quishing attacks by identifying emails or messages containing
#   QR code images where the encoded URL resolves to a recently registered
#   domain (under 30 days), a known URL shortener redirecting to a
#   credential harvesting page, or a domain impersonating postal services
#   or banking institutions. INTERPOL's 2026 Global Financial Fraud Threat
#   Assessment documents a European modus operandi where fraudsters send
#   fake QR codes claiming payment is available at a post office, routing
#   victims through a fake postal site to a cloned bank login page. In a
#   documented case, USD 110,000 was stolen through this method. The attack
#   exploits a fundamental QR code UX gap: unlike hyperlinks, QR codes do
#   not display a destination URL before the victim commits to following
#   them. The subsequent 2FA device enrollment step converts credential
#   capture into persistent account takeover, making real-time detection at
#   the email/content analysis layer critical before the victim scans.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: email_security
#   service: content_analysis
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection_qr_in_message:
#     attachment_contains_qr_code: true
#   filter_new_domain:
#     qr_destination_domain_age_days|lte: 30
#   filter_shortener_redirect:
#     qr_destination_is_shortener: true
#     redirect_final_domain_age_days|lte: 30
#   filter_brand_impersonation:
#     qr_destination_domain_impersonates:
#     - postal_service
#     - banking_institution
#     qr_destination_domain_is_legitimate: false
#   condition: 'selection_qr_in_message AND (filter_new_domain OR filter_shortener_redirect
#     OR filter_brand_impersonation)
#   
#     '
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Legitimate marketing emails containing QR codes linking to new campaign landing pages
#   - Event invitations with QR codes for registration on recently launched event platforms
#   - Payment requests from new merchants using QR-based payment systems
#   - Internal communications using QR codes linking to newly deployed internal tools
#
