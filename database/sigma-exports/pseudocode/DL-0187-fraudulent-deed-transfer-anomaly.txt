# ======================================================================
# FLAME Detection Rule: Fraudulent Deed Transfer Anomaly — Deed Recording With Document Metadata Anomalies (AI Generation Indicators) on Properties With No Prior Activity
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: high
# CFPF Phase: P4
# Threat Paths: TP-0073
#
# Description:
#   Detects deed recordings that exhibit document metadata anomalies
#   indicative of AI- generated or forged documents, particularly on
#   properties with no recent prior recording activity. The detection
#   combines document forensic signals (AI tool signatures in PDF metadata,
#   rapid creation timestamps, font inconsistencies) with property-level
#   risk factors (no prior recordings in 5 years, vacant status,
#   unencumbered, non-resident owner). Entrust reports a 40% YoY increase
#   in deepfake-related fraud attempts (2026), with AI-forged deeds,
#   government IDs, and notary stamps as primary vectors in real estate
#   title fraud. Notary commission validation is a critical check --
#   invalid or expired notary credentials are a high-confidence fraud
#   indicator.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: real_estate
#   service: deed_recordings
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection:
#     recording_type:
#     - deed_transfer
#     - quitclaim_deed
#     - warranty_deed
#     - grant_deed
#   filter_document_anomaly:
#     document_metadata_risk_score|gte: 40
#   filter_no_prior_activity:
#     prior_recordings_5yr|lte: 0
#   condition: 'selection AND (filter_document_anomaly OR filter_no_prior_activity)
#   
#     '
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Legitimate property transfers on long-held family properties with no recent activity
#   - Estate transfers after death of long-term owner
#   - Quitclaim deeds between family members for estate planning
#   - Properties in rural areas with infrequent recording activity
#   - Documents scanned or digitized from paper originals (may trigger metadata anomalies)
#
