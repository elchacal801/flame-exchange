# ======================================================================
# FLAME Detection Rule: AI-Generated Claims Document Detection — Metadata Anomalies and Generation Tool Signatures
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: high
# CFPF Phase: P4
# Threat Paths: TP-0077
#
# Description:
#   Detects AI-generated or AI-altered insurance claims documents by
#   analyzing metadata anomalies including GAN artifacts, inconsistent EXIF
#   data, and AI generation tool signatures. An estimated 25-30% of
#   insurance claims now involve some form of GenAI-altered content. This
#   rule targets the document forensics layer: photos submitted without
#   legitimate camera metadata, documents with AI tool signatures in
#   creation metadata, images with frequency domain anomalies
#   characteristic of GAN generation, and GPS coordinates that contradict
#   the claimed loss location. The insurance fraud detection market is
#   growing from USD 7.5B to USD 9.13B reflecting industry investment in
#   these countermeasures. Debener et al. (2023) demonstrate that XGBoost
#   and isolation forests provide complementary detection approaches for
#   anomalous claims.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: insurance
#   service: claims_processing
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection:
#     document_type:
#     - photo
#     - invoice
#     - medical_record
#     - repair_estimate
#     - police_report
#   filter_missing_camera:
#     exif_camera_model: null
#     document_type: photo
#   filter_ai_tool:
#     creation_tool|contains:
#     - stable diffusion
#     - midjourney
#     - dall-e
#     - comfyui
#     - automatic1111
#   filter_gan_artifacts:
#     gan_artifact_score|gte: 0.7
#   filter_frequency_anomaly:
#     frequency_domain_anomaly_score|gte: 0.8
#   filter_gps_mismatch:
#     exif_gps_loss_location_match: false
#   timeframe: 30d
#   condition: 'selection AND (filter_missing_camera OR filter_ai_tool OR filter_gan_artifacts
#     OR filter_frequency_anomaly OR filter_gps_mismatch)
#   
#     '
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Screenshots or screen captures submitted as evidence (no camera EXIF)
#   - Photos transferred via messaging apps that strip EXIF metadata
#   - Scanned documents that lose original creation metadata
#   - Legitimate use of photo editing software flagged by creation tool detection
