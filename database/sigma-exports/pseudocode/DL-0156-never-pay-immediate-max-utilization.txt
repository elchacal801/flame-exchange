# ======================================================================
# FLAME Detection Rule: Never-Pay Pattern — Immediate Max Utilization with Zero Payments
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: high
# CFPF Phase: P4
# Threat Paths: TP-0016
#
# Description:
#   Detects new credit accounts with immediate high utilization and zero
#   payment activity after 60 days — classic "never-pay" first-party fraud
#   where the consumer opens an account with no intent to pay. Unlike
#   bust-out schemes that involve a credit nurturing phase, never-pay
#   actors maximize utilization immediately after account opening and make
#   no payments whatsoever. This pattern is distinguishable from legitimate
#   new account activity because at least one billing cycle has passed with
#   zero payment response. The never-pay variant accounts for a significant
#   share of first-party fraud losses, particularly in unsecured credit
#   products.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: banking
#   service: credit_monitoring
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection:
#     account_type:
#     - consumer_credit
#     - credit_card
#   filter_new_account:
#     account_age_days|lte: 90
#   filter_high_utilization:
#     current_utilization_pct|gte: 75
#   filter_zero_payments:
#     total_payments_made: 0
#   filter_billing_cycle:
#     billing_cycles_elapsed|gte: 1
#   condition: 'selection AND filter_new_account AND filter_high_utilization AND filter_zero_payments
#     AND filter_billing_cycle
#   
#     '
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Newly opened accounts used for a single large planned purchase with payment pending
#   - Accounts opened near billing cycle boundary where payment is not yet due
#   - Corporate cards issued to employees with centralized payment processing
#
