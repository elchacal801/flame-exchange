# ======================================================================
# FLAME Detection Rule: Organized Counterfeit Goods and Non-Delivery Fraud Network — High Dispute Merchant Accounts with Drop Location Clustering and Below-Market Pricing
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: high
# CFPF Phase: P4
# Threat Paths: TP-0063
#
# Description:
#   Detects organized counterfeit goods and non-delivery fraud network
#   activity by identifying merchant accounts exhibiting multiple fraud
#   indicators simultaneously: high chargeback/dispute rates, shipping
#   addresses clustered at known drop locations, product listings priced
#   significantly below market value, and seller account age under 90 days
#   with disproportionately high transaction volume. The UNODC Organized
#   Fraud Issue Paper (2024) documents this as the most prevalent fraud
#   category by volume, with organized criminal groups operating multi-site
#   networks backed by dedicated money laundering infrastructure, recruited
#   money mules, and counterfeit supply chains. Case studies from Romania,
#   Germany, and the US show OCGs adopting formal business structures —
#   suppliers, distribution centers, online retailers — to deceive payment
#   providers and scale operations. The detection logic combines four
#   signals: (1) dispute rate exceeding 2% of transactions indicating
#   non-delivery or counterfeit complaints; (2) shipping address
#   concentration at drop locations or mail receiving services; (3) product
#   pricing 30%+ below market benchmarks; and (4) new seller accounts (<90
#   days) with anomalously high listing and transaction volume. Any two of
#   these four indicators in combination warrant investigation.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: ecommerce
#   service: merchant_monitoring
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection:
#     merchant_type:
#     - marketplace_seller
#     - ecommerce_merchant
#     - auction_seller
#   filter_high_dispute_rate:
#     dispute_rate_pct|gte: 2.0
#     dispute_count_30d|gte: 10
#   filter_drop_location:
#     shipping_address_drop_flag: true
#     shipping_address_cluster_count|gte: 5
#   filter_below_market_pricing:
#     avg_price_vs_market_ratio|lte: 0.7
#   filter_new_high_volume:
#     seller_account_age_days|lte: 90
#     transaction_count_30d|gte: 50
#   timeframe: 30d
#   condition: "selection AND (\n  (filter_high_dispute_rate AND filter_below_market_pricing)\n\
#     \  OR (filter_high_dispute_rate AND filter_new_high_volume)\n  OR (filter_drop_location\
#     \ AND filter_new_high_volume)\n  OR (filter_below_market_pricing AND filter_drop_location)\n\
#     )\n"
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Legitimate liquidation or clearance sellers offering products significantly below retail price
#   - New marketplace sellers with aggressive pricing strategy during initial launch period
#   - Fulfillment centers or warehouses that legitimately receive high volumes of shipments to a single address
#   - Seasonal sellers (holiday, event-specific) with naturally short account lifespans and high volume
