# ======================================================================
# FLAME Detection Rule: Premium SMS Billing Anomaly — Unauthorized Premium SMS Subscriptions Appearing on Accounts
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: high
# CFPF Phase: P4
# Threat Paths: TP-0072
#
# Description:
#   Detects unauthorized premium SMS subscriptions (cramming) appearing on
#   customer accounts without corresponding opt-in or consent records.
#   Premium SMS cramming generates an estimated $1.5B in unauthorized
#   consumer charges annually (CFCA/TNS 2026). Fraudsters add premium SMS
#   shortcode subscriptions to accounts through compromised billing APIs,
#   social engineering of carrier support staff, or malware-initiated WAP
#   billing triggers. The key detection signal is premium SMS subscription
#   records that lack a corresponding customer opt-in event. Multiple
#   premium subscriptions appearing on the same account within a 30-day
#   window without opt-in are high-confidence cramming indicators.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: telecom
#   service: billing
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection:
#     charge_type:
#     - premium_sms
#     - premium_content
#     - shortcode_subscription
#   filter_no_opt_in:
#     has_opt_in_record: false
#   filter_multiple_subscriptions:
#     premium_subscription_count_30d|gte: 2
#   timeframe: 30d
#   condition: selection AND filter_no_opt_in AND filter_multiple_subscriptions
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Customers who legitimately subscribed to multiple premium SMS services via web or app opt-in flows not captured in the consent management system
#   - Promotional trials that auto-convert to paid subscriptions after consent was given verbally
#   - Partner-bundled premium services included with certain plan types
#   - Consent records delayed due to system synchronization lag between opt-in platform and billing
#
