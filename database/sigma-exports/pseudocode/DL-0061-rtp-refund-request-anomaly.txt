# ======================================================================
# FLAME Detection Rule: Real-Time Payment Followed by Immediate Refund or Return Request
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: high
# CFPF Phase: P5
# Threat Paths: TP-0024
#
# Description:
#   Detects a pattern where a real-time payment (RTP or FedNow) is sent and
#   then an immediate refund, return, or Request for Return of Funds (RROF)
#   is submitted within a short window. This pattern may indicate a
#   double-dip fraud scheme where the sender initiates an irrevocable
#   payment, then exploits the refund process to recover funds while the
#   recipient also retains them.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: banking
#   service: instant_payments
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection_payment:
#     event_type: instant_payment_completed
#     payment_rail|contains:
#     - RTP
#     - FEDNOW
#   selection_refund:
#     event_type|contains:
#     - refund_request
#     - return_request
#     - rrof_submitted
#   condition: (selection_payment and selection_refund) | temporal(ordered=true, maxspan=2h,
#     by=payment_reference_id)
#   timeframe: 2h
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Legitimate duplicate payment correction where customer realizes the error immediately
#   - Merchant-initiated refund for cancelled service following immediate payment
#
# ======================================================================
# Implementation Notes:
#   - Review the condition syntax and implement using your SIEM's
#     native aggregation/correlation capabilities
#   - Splunk: Use 'stats' or 'transaction' commands
#   - Elastic: Use EQL sequences or aggregation queries
#   - Sentinel: Use KQL summarize/make-series operators
# ======================================================================
