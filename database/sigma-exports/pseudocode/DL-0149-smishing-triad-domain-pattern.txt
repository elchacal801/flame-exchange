# ======================================================================
# FLAME Detection Rule: Smishing Triad Domain Pattern and iMessage/RCS Delivery Detection
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: high
# CFPF Phase: P2
# Threat Paths: TP-0069
#
# Description:
#   Detects indicators of Smishing Triad phishing infrastructure including
#   Darcula/Magic Cat domain patterns and Chenlun kit fingerprints. The
#   Smishing Triad ecosystem has generated 194,000+ malicious domains
#   targeting users in 121+ countries, compromising 884,000+ credit cards.
#   Darcula domains follow a "com-" prefix pattern. Chenlun kits are
#   fingerprinted by the presence of ResourceRedConfig.js and
#   /ResourceConfig/urlConfig.json files. Detection also covers toll-scam
#   lure patterns (E-ZPass, SunPass, FasTrak, I-Pass) commonly used in
#   smishing campaigns.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: web_proxy
#   service: url_filtering
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection_darcula_domain:
#     url|re: ^https?://com-[a-z0-9]+
#   selection_chenlun_resource:
#     url|contains:
#     - ResourceRedConfig.js
#     - /ResourceConfig/urlConfig.json
#   selection_toll_lure:
#     url|contains:
#     - ezpass
#     - sunpass
#     - fastrak
#     - ipass
#     - toll
#     - parking-notice
#   condition: selection_darcula_domain OR selection_chenlun_resource OR selection_toll_lure
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Legitimate domains beginning with "com-" (uncommon but possible)
#   - Legitimate toll payment services (filter by known-good domains)
#   - Security researchers analyzing smishing infrastructure
#
# ======================================================================

#     native aggregation/correlation capabilities
#   - Splunk: Use 'stats' or 'transaction' commands
#   - Elastic: Use EQL sequences or aggregation queries
#   - Sentinel: Use KQL summarize/make-series operators
