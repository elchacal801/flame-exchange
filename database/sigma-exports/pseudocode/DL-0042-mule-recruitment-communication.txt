# ======================================================================
# FLAME Detection Rule: Mule Recruitment Test Deposit Pattern
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: medium
# CFPF Phase: P5
# Threat Paths: TP-0011, TP-0001
#
# Description:
#   Detects accounts receiving multiple small "test" deposits from many
#   distinct external sources. Mule recruiters validate new mule accounts
#   by sending micro-deposits from various fraud proceeds before routing
#   larger amounts. The pattern of many small inbound transfers from
#   unrelated senders is highly anomalous for personal accounts.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: banking
#   service: ach_processing
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection_small_deposits:
#     event_type:
#     - ACH_CREDIT
#     - P2P_CREDIT
#     - WIRE_CREDIT
#     transaction_amount|lte: 100
#     transaction_amount|gte: 1
#   timeframe: 14d
#   condition: selection_small_deposits | count(distinct originator_id) by account_id
#     > 8
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Account verification micro-deposits from legitimate fintech platforms linking accounts
#   - Social media influencers receiving many small tips or donations
#
# ======================================================================
# Implementation Notes:
#   - Review the condition syntax and implement using your SIEM's
#     native aggregation/correlation capabilities
#   - Splunk: Use 'stats' or 'transaction' commands
#   - Elastic: Use EQL sequences or aggregation queries
#   - Sentinel: Use KQL summarize/make-series operators
