# ======================================================================
# FLAME Detection Rule: Wangiri Callback Velocity — High Volume of Short-Duration Outbound International Calls From Single Source
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: high
# CFPF Phase: P4
# Threat Paths: TP-0071
#
# Description:
#   Detects Wangiri ("one-ring") fraud patterns where a high volume of
#   short-duration (< 5 seconds) outbound international calls originate
#   from a single source, indicating subscriber callbacks to premium-rate
#   numbers. In a Wangiri scheme, fraudsters place brief inbound calls from
#   premium-rate numbers to trigger victim callbacks. The callbacks connect
#   to IVR systems or premium content lines that accumulate per-minute
#   charges. This rule detects the callback side of the attack: clusters of
#   brief outbound international calls from individual subscribers or PBX
#   systems to premium-rate destinations. Average victim loss is $15-$50
#   per callback; aggregate campaign revenues reach $100K-$1M
#   (Subex/Akamai).
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: telecom
#   service: call_detail_records
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection:
#     call_direction: outbound
#     call_type: international
#   filter_short_duration:
#     call_duration_seconds|lt: 5
#   filter_volume:
#     short_call_count_24h|gte: 10
#   filter_destination_diversity:
#     unique_premium_destinations_24h|gte: 3
#   timeframe: 24h
#   condition: 'selection AND filter_short_duration AND filter_volume AND filter_destination_diversity
#   
#     '
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - International business call attempts with connection failures (busy/no answer)
#   - Automated system health checks or monitoring calls to international endpoints
#   - Call center operations with short-duration international callback attempts
#   - Network testing or carrier quality assurance calls
#
