# ======================================================================
# FLAME Detection Rule: Bot Traffic Behavioral Anomaly — Mechanical Consistency in Affiliate Session Metrics
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: high
# CFPF Phase: P4
# Threat Paths: TP-0076
#
# Description:
#   Detects bot-generated affiliate traffic by identifying sessions with
#   mechanical behavioral consistency — identical session durations, linear
#   scroll patterns, and zero mouse movement variance — that indicate
#   automated rather than human browsing. Bot networks account for
#   approximately 40% of all click fraud, and AI-powered bots increasingly
#   mimic individual human sessions. However, bot traffic still exhibits
#   statistically detectable patterns at the population level: genuine
#   human traffic has natural variance in behavioral metrics, while bot
#   traffic — even sophisticated AI bots — produces distributions with
#   anomalously low standard deviation. This rule analyzes session-level
#   behavioral distributions per affiliate to detect mechanical consistency
#   at the aggregate level.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: advertising
#   service: web_analytics
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection:
#     traffic_source:
#     - affiliate
#   filter_volume:
#     session_count|gte: 100
#   filter_mechanical:
#     session_duration_stddev|lt: 1.0
#   filter_scroll:
#     scroll_depth_stddev|lt: 2.0
#   filter_mouse:
#     zero_mouse_ratio|gte: 0.5
#   timeframe: 7d
#   condition: 'selection AND filter_volume AND (filter_mechanical OR filter_scroll OR
#     filter_mouse)
#   
#     '
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Legitimate automated testing or monitoring tools generating uniform traffic patterns
#   - Accessibility tools that produce consistent browsing patterns (screen readers, keyboard-only navigation)
#   - Embedded browser views in mobile apps with constrained interaction patterns
#

#     native aggregation/correlation capabilities

#   - Elastic: Use EQL sequences or aggregation queries
#   - Sentinel: Use KQL summarize/make-series operators
