# Level: critical

# Threat Paths: TP-0029, TP-0018

# Description:
#   Detects repeated liveness check failures from the same device followed
#   by eventual success — indicates iterative deepfake tuning where
#   fraudsters adjust their deepfake parameters until they pass liveness
#   detection. Normal users fail once (bad lighting) and succeed quickly;
#   attackers show a pattern of 3+ failures with improving scores.
#   Real-time face-swap applications and pre-recorded deepfake videos are
#   used to defeat challenges requiring head turns, blinking, or phrase
#   repetition. The iterative improvement pattern — each attempt scoring
#   higher than the last — is a strong indicator of adversarial tuning
#   against the liveness detection model.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: banking
#   service: document_verification
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection_liveness_check:
#     event_type: LIVENESS_CHECK
#   filter_failure_pattern:
#     device_fingerprint|exists: true
#   condition: 'selection_liveness_check AND filter_failure_pattern AND (count(liveness_result
#     = "fail" WHERE same device_fingerprint within 24h) >= 3 AND liveness_result = "pass"
#     within same 24h window) OR (liveness_score shows monotonic increase across 3+ consecutive
#     attempts from same device_fingerprint)
#   
#     '
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Users with poor camera quality or lighting requiring multiple genuine attempts
#   - Users unfamiliar with liveness check process who need several tries
#   - Network connectivity issues causing liveness check session interruptions
