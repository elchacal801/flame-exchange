# ======================================================================
# FLAME Detection Rule: SSN Age and Issuance Mismatch — Synthetic Identity Indicator
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: high
# CFPF Phase: P2
# Threat Paths: TP-0003, TP-0029
#
# Description:
#   Detects credit applications where the SSN characteristics are
#   inconsistent with the applicant's stated identity — a primary indicator
#   of synthetic identity fabrication. Key signals: (1) SSN issued after
#   the applicant's date of birth suggests the SSN belongs to someone else
#   (minors, immigrants, recently deceased), (2) post-2011 randomized SSNs
#   that don't match SSA area number geography, (3) SSN first appearing at
#   credit bureaus within the last 18 months combined with an applicant age
#   > 25 (natural credit profiles start appearing at 18-22). The eCBSV
#   (electronic Consent Based SSN Verification) service can confirm
#   SSN-name-DOB matches, but adoption is inconsistent.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: banking
#   service: account_onboarding
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection_application:
#     event_type: CREDIT_APPLICATION
#   filter_ssn_age_mismatch:
#     ssn_issue_year_estimated|gt: 0
#   condition: 'selection_application AND (applicant_stated_age > 25 AND credit_file_age_months
#     < 18) OR (ssn_issue_year_estimated > applicant_birth_year) OR (applicant_stated_age
#     < 16 AND credit_inquiry_exists = true)
#   
#     '
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Recent immigrants who obtained SSNs later in life (legitimate late credit file)
#   - Young adults with limited credit history applying for first credit products
#   - Data entry errors in date of birth fields
#
