# ======================================================================
# FLAME Detection Rule: Post-AiTM Inbox Rule Manipulation — New Rules from Risky Sign-In
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: high
# CFPF Phase: P3
# Threat Paths: TP-0067
#
# Description:
#   Detects inbox rule creation or modification in Exchange Online
#   following a sign-in flagged with medium or high risk, or following
#   authentication from a previously unseen IP address. Post-AiTM inbox
#   rule manipulation is a characteristic Phase 3 (Positioning) technique:
#   within minutes of session hijack, attackers create inbox rules to
#   redirect or delete security alerts, payment notifications, and
#   colleague communications. This detection correlates Exchange Online
#   Unified Audit Log operations (New-InboxRule, Set-InboxRule, Set-Mailbox
#   forwarding) with Entra ID risky sign-in events.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: m365
#   service: exchange
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection_inbox_rule:
#     Operation:
#     - New-InboxRule
#     - Set-InboxRule
#     - Enable-InboxRule
#     - Set-Mailbox
#   selection_suspicious_params:
#     Parameters|contains:
#     - ForwardTo
#     - ForwardAsAttachmentTo
#     - RedirectTo
#     - DeleteMessage
#     - MarkAsRead
#     - MoveToFolder
#   correlation:
#     type: temporal
#     description: 'Inbox rule creation with suspicious parameters must occur within 24h
#       of a sign-in flagged with medium or high risk (RiskLevelDuringSignIn or RiskLevelAggregated).
#       Requires cross-log correlation between Exchange Online audit logs and Entra ID
#       sign-in logs.
#   
#       '
#   timeframe: 24h
#   condition: 'selection_inbox_rule AND selection_suspicious_params AND user had medium/high
#     risk sign-in within preceding 24 hours
#   
#     '
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Legitimate inbox rule creation by users shortly after a sign-in flagged as risky due to VPN or travel
#   - IT administrators creating mail flow rules from new IP addresses
#   - Automated email management tools creating rules on behalf of users
#
# ======================================================================
# Implementation Notes:
#   - Review the condition syntax and implement using your SIEM's
#     native aggregation/correlation capabilities
#   - Splunk: Use 'stats' or 'transaction' commands
#   - Elastic: Use EQL sequences or aggregation queries
#   - Sentinel: Use KQL summarize/make-series operators
