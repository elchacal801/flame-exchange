# ======================================================================
# FLAME Detection Rule: First-Party Credit Buildup and Sudden Spike
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: high
# CFPF Phase: P4
# Threat Paths: TP-0016
#
# Description:
#   Detects consumer accounts following a first-party fraud pattern: steady
#   good payments for 6-12 months then sudden utilization spike and missed
#   payments within 60 days. Unlike synthetic identity (TP-0003), the
#   identity is real but the intent is fraudulent. First-party bust-out
#   actors nurture credit profiles with consistent on-time payments and low
#   utilization, then rapidly maximize available credit before abandoning
#   the account. The transition from responsible usage to aggressive
#   drawdown is the key behavioral signal. LexisNexis estimates first-party
#   fraud now represents 36% of all detected fraud events globally.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: banking
#   service: credit_monitoring
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection:
#     account_type:
#     - consumer_credit
#     - credit_card
#   filter_good_history:
#     on_time_payment_months|gte: 6
#   filter_utilization_spike:
#     previous_avg_utilization_pct|lt: 40
#     current_utilization_pct|gte: 85
#   filter_missed_payment:
#     days_since_first_missed_payment|lte: 60
#     days_since_utilization_spike|lte: 60
#   timeframe: 60d
#   condition: 'selection AND filter_good_history AND filter_utilization_spike AND filter_missed_payment
#   
#     '
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Legitimate financial hardship (job loss, medical emergency) causing simultaneous high spending and missed payments
#   - Consumers running up balances during planned bankruptcy filing
#   - Balance transfer activity creating temporary utilization spike on source account
#

#     native aggregation/correlation capabilities

#   - Elastic: Use EQL sequences or aggregation queries
#   - Sentinel: Use KQL summarize/make-series operators
