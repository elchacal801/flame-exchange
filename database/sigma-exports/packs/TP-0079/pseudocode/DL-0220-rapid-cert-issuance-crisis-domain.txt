# ======================================================================
# FLAME Detection Rule: Rapid Certificate Issuance on Crisis-Themed Domain — Sub-48h Registration-to-Cert Velocity
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: high
# CFPF Phase: P2
# Threat Paths: TP-0086, TP-0079
#
# Description:
#   Detects TLS certificate issuance within 48 hours of domain registration
#   on crisis-themed domains. Legitimate organizations typically register
#   domains days to weeks before certificate provisioning, while
#   crisis-exploitation infrastructure shows a characteristic sub-48h
#   registration-to-cert velocity as threat actors race to deploy
#   HTTPS-credible phishing pages before awareness develops. Certificate
#   transparency logs provide the data source. CrimsonVector documented
#   this rapid-cert pattern across 1,435 crisis-themed domains during the
#   2026 oil shock, with Let's Encrypt and ZeroSSL as the primary issuers.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: banking
#   service: domain_intelligence
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection_crisis_keyword:
#     domain_name|contains:
#     - subsidy
#     - relief
#     - assistance
#     - fuel-
#     - petrol-
#     - energy-emergency
#     - oil-invest
#     - crude-trad
#     - commodity-trad
#   filter_rapid_cert:
#     hours_registration_to_cert|lte: 48
#   filter_free_ca:
#     certificate_issuer|contains:
#     - Let's Encrypt
#     - ZeroSSL
#     - Buypass
#   condition: selection_crisis_keyword AND filter_rapid_cert AND filter_free_ca
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Legitimate rapid-deployment sites using automated certificate provisioning
#   - CDN-provisioned certificates on legitimate customer domains
#
