# ======================================================================
# FLAME Detection Rule: Carrier Change Authentication Correlation
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: critical
# CFPF Phase: P2
# Threat Paths: TP-0008, TP-0001
#
# Description:
#   Detects a carrier or SIM change event immediately followed by an
#   authentication attempt on a banking platform. SIM swap attacks involve
#   social engineering or insider compromise at a mobile carrier to port
#   the victim's phone number to an attacker-controlled SIM, after which
#   the attacker uses SMS-based MFA to access accounts. The tight temporal
#   correlation between carrier change and login is the key indicator.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: telecom
#   service: identity_management
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection_carrier_change:
#     event_type|contains:
#     - sim_swap
#     - carrier_change
#     - number_port
#     status: completed
#   selection_auth:
#     event_type: login_attempt
#     mfa_method: sms
#     mfa_result: success
#   timeframe: 120m
#   condition: 'selection_carrier_change | followed_by selection_auth on phone_number
#     within 120m
#   
#     '
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Legitimate users who switch carriers and immediately log in to their banking apps
#   - Planned device upgrades where users authenticate on new hardware right after activation
#
# ======================================================================
# Implementation Notes:
#   - Review the condition syntax and implement using your SIEM's
#     native aggregation/correlation capabilities
#   - Splunk: Use 'stats' or 'transaction' commands
#   - Elastic: Use EQL sequences or aggregation queries
#   - Sentinel: Use KQL summarize/make-series operators
# ======================================================================
