# SPL/Lucene/KQL by pySigma backends.
#
# Level: high
# CFPF Phase: P2
# Threat Paths: TP-0008
#
# Description:
#   Detects potential SIM swap attacks by querying network-level APIs
#   before MFA code delivery. SIM swap fraud surged 1,055% in the UK (3,000
#   cases in 2024). Three network-level APIs provide the most effective
#   detection: SIM Swap Check (has the SIM been changed recently?), Number
#   Verification (is the number still assigned to the expected device?),
#   and Call Forwarding Signal (is call forwarding active?). Correlating
#   positive signals from these APIs with subsequent MFA authentication
#   attempts provides high-confidence SIM swap detection.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: telecom
#   service: number_intelligence
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection:
#     api_type:
#     - sim_swap_check
#     - number_verification
#     - call_forwarding_signal
#   condition: 'sim_swap_check.recently_swapped == true OR number_verification.device_mismatch
#     == true OR call_forwarding_signal.forwarding_active == true
#   
#     '
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Legitimate SIM replacements (lost/damaged phone)
#   - Carrier-initiated SIM migrations (network upgrades)
#   - Users activating call forwarding for legitimate reasons
#
