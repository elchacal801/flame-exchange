# ======================================================================
# FLAME Detection Rule: Post-Compromise Inbox Rule Manipulation — Rule Creation After New-IP Authentication
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: high
# CFPF Phase: P3
# Threat Paths: TP-0067, TP-0002
#
# Description:
#   Detects inbox rule creation or modification within 24 hours of a
#   sign-in from a previously unseen IP address. AiTM phishing kit
#   operators create inbox rules within minutes of session hijack to
#   redirect payment notifications, delete security alerts, and intercept
#   colleague communications. Rules forwarding to external domains or
#   deleting emails matching specific sender/subject patterns are
#   high-confidence indicators of post-compromise activity.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: m365
#   service: exchange
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection_inbox_rule:
#     Operation:
#     - New-InboxRule
#     - Set-InboxRule
#     - Enable-InboxRule
#   correlation:
#     type: temporal
#     description: 'Inbox rule creation must occur within 24h of a sign-in from an IP
#       address not previously seen for this user (first-seen within 24h window). Requires
#       cross-log correlation between Exchange Online audit logs and Entra ID sign-in
#       logs.
#   
#       '
#   timeframe: 24h
#   condition: 'selection_inbox_rule AND user authenticated from previously-unseen IP
#     within preceding 24 hours
#   
#     '
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Users creating legitimate inbox rules from new locations (travel, new office)
#   - IT administrators configuring inbox rules remotely
#   - Automated rule creation by legitimate email management tools
#
# ======================================================================
# Implementation Notes:
#   - Review the condition syntax and implement using your SIEM's
#     native aggregation/correlation capabilities
#   - Splunk: Use 'stats' or 'transaction' commands
#   - Elastic: Use EQL sequences or aggregation queries
#   - Sentinel: Use KQL summarize/make-series operators
# ======================================================================
