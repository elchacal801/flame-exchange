# ======================================================================
# FLAME Detection Rule: Crypto Off-Ramp After Wire or ACH Receipt
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: high
# CFPF Phase: P5
# Threat Paths: TP-0022, TP-0002
#
# Description:
#   Detects a pattern where an account receives a wire or ACH credit and
#   then promptly initiates a transfer to a known cryptocurrency exchange
#   or digital asset platform. Converting fraud proceeds to cryptocurrency
#   provides near-irreversible laundering, making this a critical
#   monetization indicator.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: banking
#   service: transaction_monitoring
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection_inbound:
#     event_type:
#     - WIRE_CREDIT
#     - ACH_CREDIT
#     transaction_amount|gte: 1000
#   selection_crypto_outbound:
#     event_type:
#     - WIRE_INITIATION
#     - ACH_DEBIT
#     beneficiary_category: crypto_exchange
#     beneficiary_name|contains:
#     - Coinbase
#     - Binance
#     - Kraken
#     - FTX
#     - Gemini
#     - Crypto.com
#     - Bitfinex
#   timeframe: 24h
#   condition: selection_inbound | near selection_crypto_outbound by account_id
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Legitimate cryptocurrency investors or traders funding exchange accounts from bank proceeds
#   - Businesses with crypto treasury strategies converting fiat after receivables
#
# ======================================================================
# Implementation Notes:
#   - Review the condition syntax and implement using your SIEM's
#     native aggregation/correlation capabilities
#   - Splunk: Use 'stats' or 'transaction' commands
#   - Elastic: Use EQL sequences or aggregation queries
#   - Sentinel: Use KQL summarize/make-series operators
# ======================================================================
