# ======================================================================
# FLAME Detection Rule: Peer-to-Peer Transfer Chain (Smurfing Pattern)
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: medium
# CFPF Phase: P5
# Threat Paths: TP-0011, TP-0022, TP-0002
#
# Description:
#   Detects chains of peer-to-peer transfers across multiple accounts where
#   funds move through a sequence of intermediary accounts (A to B to C to
#   D). This smurfing pattern is used to layer fraud proceeds through
#   multiple hops, making tracing difficult.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: banking
#   service: transaction_monitoring
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection_p2p:
#     event_type:
#     - P2P_TRANSFER
#     - INTERNAL_TRANSFER
#     - ZELLE_SEND
#     - VENMO_TRANSFER
#     transaction_amount|gte: 500
#   selection_chain:
#     inbound_event: P2P_CREDIT
#     outbound_event: P2P_DEBIT
#   timeframe: 2h
#   condition: 'selection_p2p | count() by account_id > 3 and selection_chain | count(distinct
#     counterparty_account_id) by account_id > 2
#   
#     '
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Group event organizers collecting and redistributing shared expenses
#   - Family members transferring funds between accounts for household management
#
# ======================================================================
# Implementation Notes:
#   - Review the condition syntax and implement using your SIEM's
#     native aggregation/correlation capabilities
#   - Splunk: Use 'stats' or 'transaction' commands
#   - Elastic: Use EQL sequences or aggregation queries
#   - Sentinel: Use KQL summarize/make-series operators
# ======================================================================
