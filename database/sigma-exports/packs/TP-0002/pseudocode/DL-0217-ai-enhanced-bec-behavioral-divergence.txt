# ======================================================================
# FLAME Detection Rule: AI-Enhanced BEC — Payment Request Behavioral Divergence from Sender Baseline
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: medium
# CFPF Phase: P3
# Threat Paths: TP-0002, TP-0007
#
# Description:
#   Detects AI-enhanced business email compromise by identifying wire
#   transfer requests where the communication pattern diverges
#   significantly from the sender's established baseline. FBI IC3 2025
#   reported $30 million in losses from AI-enabled BEC where chat
#   generators created official-sounding emails mimicking company
#   executives and voice cloning was used for wire authorization requests.
#   AI-generated BEC content, while linguistically fluent, often lacks the
#   sender's authentic communication patterns — writing style, formality
#   level, time-of-day preferences, typical request amounts, and approval
#   workflows. This rule targets Phase 3 (Positioning) where the fraudulent
#   request is crafted and delivered, before the wire transfer is executed.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: banking
#   service: transaction_monitoring
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection:
#     request_type: wire_transfer_request
#     request_channel: email
#   filter_behavioral_divergence:
#     sender_style_divergence_score|gte: 2.0
#   filter_amount_anomaly:
#     request_amount_vs_sender_baseline_pct|gte: 200
#   filter_timing_anomaly:
#     request_outside_sender_typical_hours: true
#   condition: 'selection AND (filter_behavioral_divergence OR (filter_amount_anomaly
#     AND filter_timing_anomaly))
#   
#     '
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Executives writing under time pressure with atypical brevity or urgency
#   - Emails sent by executive assistants on behalf of the named sender
#   - Automated or template-based payment requests that differ from freeform communication style
#   - Cross-timezone communications appearing outside normal hours
#
