# ======================================================================
# FLAME Detection Rule: ACH Return Rate Anomaly
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: high
# CFPF Phase: P5
# Threat Paths: TP-0002, TP-0001
#
# Description:
#   Detects ACH originators with an elevated return or reversal rate
#   exceeding normal thresholds. High ACH return rates (especially R10
#   unauthorized, R29 corporate not authorized) may indicate the account is
#   being used to initiate unauthorized debits against victim accounts.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: banking
#   service: ach_processing
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection_returns:
#     event_type: ACH_RETURN
#     return_code:
#     - R10
#     - R29
#     - R07
#     - R08
#   timeframe: 30d
#   condition: selection_returns | count() by originator_id > 5
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Legitimate businesses with high dispute rates due to subscription billing models
#   - Payroll processors during seasonal employment transitions
#
# ======================================================================
# Implementation Notes:
#   - Review the condition syntax and implement using your SIEM's
#     native aggregation/correlation capabilities
#   - Splunk: Use 'stats' or 'transaction' commands
#   - Elastic: Use EQL sequences or aggregation queries
#   - Sentinel: Use KQL summarize/make-series operators
# ======================================================================
