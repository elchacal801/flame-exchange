#
# Level: high
# CFPF Phase: P2
# Threat Paths: TP-0024, TP-0012
#
# Description:
#   Detects multiple failed Zelle recipient lookups from a single account
#   or session, which may indicate enumeration activity. Attackers probe
#   email addresses and phone numbers to identify which are registered with
#   Zelle before launching targeted social engineering or authorized push
#   payment attacks against confirmed users.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: banking
#   service: a2a_processing
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection:
#     event_type: zelle_recipient_lookup
#     lookup_result: not_found
#   condition: selection | count() by account_id > 10
#   timeframe: 1h
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - User trying to send money to friends who have not yet enrolled in Zelle
#   - Administrative staff searching for payees across a new vendor list
