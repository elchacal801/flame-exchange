# FLAME Detection Rule: Employment Fraud Metadata Feature Detection
# ======================================================================

# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: high
# CFPF Phase: P2
# Threat Paths: TP-0015
#
# Description:
#   Detects fraudulent job postings using metadata features from the EMSCAD
#   benchmark (18,000 samples). Key finding: when both telecommuting and
#   company logo flags equal zero, there is a 92% probability of fraud.
#   Geographic concentration patterns (e.g., Bakersfield 15:1 fake-to-real
#   ratio) provide additional signal. DistilBERT+SMOTE models achieve
#   strong results; DNNs reach ~98% accuracy. This rule implements the
#   high-confidence metadata feature checks that require no NLP processing.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: job_platform
#   service: posting_analysis
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection:
#     posting_type: job_listing
#   condition: '(telecommuting == 0 AND has_company_logo == 0) OR (fraud_geographic_score
#     > 0.8) OR (salary_experience_ratio > 3.0 AND has_company_logo == 0)
#   
#     '
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Small businesses or startups without established branding (no logo)
#   - Legitimate remote work listings that don't flag telecommuting
#   - Job postings in geographically concentrated industries
#
