# ======================================================================
# FLAME Detection Rule: Bulletproof Hosting ASN Connection Correlation — Traffic to Known BPH Networks
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: medium
# CFPF Phase: P1
# Threat Paths: TP-0048, TP-0061
#
# Description:
#   Detects DNS resolution or network connections to IP addresses within
#   known bulletproof hosting (BPH) Autonomous System Number (ASN) ranges,
#   domains registered via the NiceNIC registrar, or domains using Dynamic
#   DNS providers such as afraid.org. Silent Push (2025) documented 200+
#   BPH providers controlling approximately 1 million IP addresses
#   globally. This rule correlates network traffic against a curated list
#   of BPH ASNs identified through abuse pattern analysis, including
#   providers with high DGA domain density, disposable abuse contact
#   emails, shell company registrations in permissive jurisdictions, and
#   self-declared bulletproof hosting marketing. Additionally, the rule
#   flags domains registered through NiceNIC — a registrar favored by
#   threat actors (including Scattered Spider and The Com) due to its
#   requirement for a Power of Attorney to process takedown requests — and
#   domains leveraging Dynamic DNS services that create BPH-like
#   abuse-resistant infrastructure. Any match warrants investigation for
#   potential fraud-enabling infrastructure, C2 communication, or criminal
#   hosting activity.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: network
#   service: dns_monitoring
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection_bph_asn:
#     destination_asn:
#     - AS152194
#     - AS214351
#     - AS213194
#     - AS214943
#     - AS34985
#     - AS48589
#     - AS49217
#     - AS214940
#     - AS140224
#     - AS200019
#     - AS49042
#     - AS45839
#   selection_nicenic_registrar:
#     domain_registrar|contains: nicenic
#   selection_dynamic_dns:
#     query|endswith:
#     - .afraid.org
#     - .mooo.com
#     - .hopto.org
#     - .zapto.org
#     - .sytes.net
#   condition: selection_bph_asn OR selection_nicenic_registrar OR selection_dynamic_dns
#   timeframe: 24h
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Legitimate websites incidentally hosted on BPH ASNs
#   - Security researchers investigating BPH infrastructure
