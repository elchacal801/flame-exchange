# ======================================================================
# FLAME Detection Rule: Crypto Token Rug Pull Early Warning — On-Chain and OSINT Indicators
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: high
# CFPF Phase: P4
# Threat Paths: TP-0017, TP-0049
#
# Description:
#   Detects cryptocurrency rug pull / exit scam indicators using on-chain
#   and OSINT signals. Rug pulls constituted 68% of all crypto scams in
#   2025 (350+ cases 2024). On-chain indicators: Uniswap liquidity removal
#   >80% in single transaction, token holder concentration (top 5 holders
#   >60%), contract functions with hidden mint/pause/ blacklist
#   capabilities. OSINT indicators: social media silence after promotion
#   burst, developer wallet activity cessation, Telegram/Discord channel
#   deletion. Critical finding: OSINT signals precede on-chain anomalies by
#   days or weeks.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: blockchain
#   service: defi_monitoring
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection_onchain:
#     event_type:
#     - liquidity_removal
#     - token_transfer
#     - contract_interaction
#   condition: 'liquidity_removal_pct > 80 in single tx OR top5_holder_concentration >
#     0.60 OR contract_has_hidden_functions == true
#   
#     '
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Legitimate liquidity rebalancing by project teams
#   - Token migrations requiring liquidity withdrawal
#   - Early-stage tokens with naturally concentrated holdings
#
