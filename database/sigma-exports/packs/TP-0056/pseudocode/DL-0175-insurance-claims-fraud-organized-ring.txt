# ======================================================================
# FLAME Detection Rule: Organized Insurance Claims Fraud — Shared Professional Enablers and Claim Timing Clustering Around Policy Inception
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: high
# CFPF Phase: P4
# Threat Paths: TP-0056
#
# Description:
#   Detects organized insurance claims fraud patterns where the same
#   claimant or provider appears across multiple claims with shared
#   professional enablers (medical provider, legal representative, repair
#   shop), or where claim timing is clustered around policy inception.
#   INTERPOL's 2026 Global Financial Fraud Threat Assessment identifies
#   insurance fraud — notably car and medical insurance claims — as
#   escalating globally, with Europe most impacted: the UK recorded a 25%
#   spike in motor insurance fraud in early 2025, France detected nearly
#   USD 1 billion in fraud in 2024, and German insurers flagged 10% of
#   claims as suspicious with annual losses exceeding USD 7 billion. The
#   detection logic targets two primary signatures: (1) network analysis
#   identifying claimants, providers, legal representatives, or repair
#   shops appearing across multiple claims — indicating an organized fraud
#   ring with division of labor; and (2) temporal analysis identifying
#   claims filed within 90 days of policy inception with high severity,
#   which indicates opportunistic application fraud followed by immediate
#   fraudulent claim filing. The combination of shared enablers and
#   inception-proximate timing is a strong indicator of organized ring
#   activity.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: insurance
#   service: claims_management
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection:
#     claim_type:
#     - motor
#     - medical
#     - personal_injury
#   filter_shared_enablers:
#     shared_provider_across_claims|gte: 2
#     shared_legal_rep_across_claims|gte: 2
#   filter_claimant_recurrence:
#     claimant_claim_count_24m|gte: 3
#   filter_inception_proximity:
#     days_from_inception_to_claim|lte: 90
#     claim_severity:
#     - high
#     - total_loss
#   timeframe: 24m
#   condition: 'selection AND (filter_shared_enablers OR filter_claimant_recurrence OR
#     filter_inception_proximity)
#   
#     '
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - High-volume legitimate medical providers (emergency rooms, urgent care clinics) naturally appearing across many claims
#   - Law firms specializing in personal injury that legitimately represent many claimants
#   - Legitimate repair shops in areas with high accident rates appearing across multiple motor claims
#   - New policyholders who experience genuine accidents shortly after policy inception
#
