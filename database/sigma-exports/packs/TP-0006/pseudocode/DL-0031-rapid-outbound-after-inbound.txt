# ======================================================================
# FLAME Detection Rule: Rapid Outbound Wire After Large Inbound
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: high
# CFPF Phase: P4
# Threat Paths: TP-0001, TP-0002, TP-0006
#
# Description:
#   Detects a large outbound wire or ACH transfer initiated within hours of
#   receiving a large inbound deposit. This pass-through pattern is
#   characteristic of compromised accounts being used as intermediary
#   conduits in fraud or money laundering schemes.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: banking
#   service: transaction_monitoring
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection_inbound:
#     event_type:
#     - WIRE_CREDIT
#     - ACH_CREDIT
#     transaction_amount|gte: 10000
#   selection_outbound:
#     event_type:
#     - WIRE_INITIATION
#     - ACH_DEBIT
#     transaction_amount|gte: 8000
#   timeframe: 6h
#   condition: selection_inbound | near selection_outbound by account_id
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Real estate escrow closings with same-day receipt and disbursement
#   - Business accounts receiving client payments and immediately paying suppliers
#
# ======================================================================
# Implementation Notes:
#   - Review the condition syntax and implement using your SIEM's
#     native aggregation/correlation capabilities
#   - Splunk: Use 'stats' or 'transaction' commands
#   - Elastic: Use EQL sequences or aggregation queries
#   - Sentinel: Use KQL summarize/make-series operators
# ======================================================================
