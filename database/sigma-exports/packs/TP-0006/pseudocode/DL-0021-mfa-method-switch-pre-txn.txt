# ======================================================================
# FLAME Detection Rule: MFA Method Switch Before High-Value Transaction
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: high
# CFPF Phase: P3
# Threat Paths: TP-0001, TP-0005, TP-0006
#
# Description:
#   Detects when an account's multi-factor authentication method is changed
#   (e.g., from hardware token to SMS or email OTP) shortly before a
#   high-value wire or ACH transaction is initiated. Attackers downgrade
#   MFA to a method they control prior to executing fraud.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: banking
#   service: identity_management
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection_mfa_change:
#     event_type: MFA_METHOD_CHANGE
#     new_mfa_method:
#     - sms_otp
#     - email_otp
#     - voice_call
#   selection_high_value_txn:
#     event_type:
#     - WIRE_INITIATION
#     - ACH_DEBIT
#     transaction_amount|gte: 5000
#   timeframe: 4h
#   condition: selection_mfa_change | near selection_high_value_txn by account_id
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Customer who lost their hardware token and is making a legitimate scheduled payment
#   - Planned MFA migration events by IT or security teams
#
# ======================================================================
# Implementation Notes:
#   - Review the condition syntax and implement using your SIEM's
#     native aggregation/correlation capabilities
#   - Splunk: Use 'stats' or 'transaction' commands
#   - Elastic: Use EQL sequences or aggregation queries
#   - Sentinel: Use KQL summarize/make-series operators
# ======================================================================
