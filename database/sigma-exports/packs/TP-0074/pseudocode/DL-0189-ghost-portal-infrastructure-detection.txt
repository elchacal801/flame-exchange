# ======================================================================
# FLAME Detection Rule: Ghost Portal Infrastructure Detection — Fake Insurance Verification on Recently Registered Domains
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: high
# CFPF Phase: P3
# Threat Paths: TP-0074
#
# Description:
#   Detects ghost portal infrastructure — fraudulent insurance verification
#   websites deployed on recently registered domains that return
#   valid-looking but unverifiable policy numbers. Ghost portals represent
#   the evolution of ghost broking into a fraud-as-a-service model, where
#   operators build and sell fake insurance verification platforms to other
#   fraudsters. These portals are typically deployed on domains registered
#   within the past 90 days, use insurance-related keywords, and mimic
#   legitimate insurer verification workflows. Detection focuses on DNS
#   queries and web traffic to recently registered domains containing
#   insurance verification terminology.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: network
#   service: dns_proxy
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection:
#     query|contains:
#     - insurance-verify
#     - policy-check
#     - insurance-confirm
#     - motor-verify
#     - cover-check
#     - insure-validate
#   filter_domain_age:
#     domain_age_days|lt: 90
#   filter_volume:
#     query_count|gte: 10
#   timeframe: 7d
#   condition: 'selection AND filter_domain_age AND filter_volume
#   
#     '
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Newly launched legitimate insurance comparison or verification services
#   - Insurance technology startups with recently registered domains
#   - Internal testing of insurance verification APIs on new domains
#
