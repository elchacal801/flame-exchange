# ======================================================================
# FLAME Detection Rule: Telecom Subscription Velocity — Multiple New Service Activations From Same Identity/Device Within 30 Days
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: high
# CFPF Phase: P3
# Threat Paths: TP-0072, TP-0003
#
# Description:
#   Detects subscription fraud patterns where the same identity or device
#   IMEI is used to activate multiple new telecom service accounts within a
#   30-day window. Subscription fraud is the top concern for 53% of telecom
#   executives (TransUnion 2025). Fraudsters use synthetic identities
#   (TP-0003) with GenAI-generated KYC documents to open accounts across
#   multiple carriers, acquiring subsidized flagship devices on installment
#   plans with no payment intent. The multi-carrier activation pattern is
#   the strongest detection signal, as legitimate customers rarely activate
#   3+ accounts across 2+ carriers in a 30-day period. Devices acquired
#   through subscription fraud are resold at 60-80% of retail value on
#   secondary markets.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: telecom
#   service: account_activations
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection:
#     activation_type: new_account
#   filter_velocity:
#     activation_count_30d|gte: 3
#   filter_multi_carrier:
#     unique_carriers_30d|gte: 2
#   timeframe: 30d
#   condition: selection AND (filter_velocity OR filter_multi_carrier)
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Corporate accounts activating multiple lines for new employees
#   - Legitimate device upgrades across carrier switches during promotional periods
#   - Family plan activations where multiple members onboard simultaneously
#   - Resellers and authorized dealers activating demonstration or inventory devices
#
