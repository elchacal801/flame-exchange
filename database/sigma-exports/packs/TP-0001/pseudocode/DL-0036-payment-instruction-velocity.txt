# ======================================================================
# FLAME Detection Rule: Payment Instruction Velocity Spike
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: high
# CFPF Phase: P4
# Threat Paths: TP-0001, TP-0006
#
# Description:
#   Detects when multiple payment instructions (wires, ACH batches, or
#   internal transfers) are submitted in rapid succession from a single
#   account or user session. High-velocity payment submissions suggest
#   automated or scripted fraud execution, often seen after credential
#   compromise.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: banking
#   service: transaction_monitoring
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection:
#     event_type:
#     - WIRE_INITIATION
#     - ACH_BATCH_SUBMIT
#     - INTERNAL_TRANSFER
#     - PAYMENT_INSTRUCTION
#   timeframe: 30m
#   condition: selection | count() by session_id > 5
#   enrichment:
#     aggregate: sum(transaction_amount) by account_id
#     threshold_multiplier: 3.0
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Corporate treasury operations submitting batch payments during end-of-day processing
#   - Payroll administrators processing multi-entity payroll batches
#
# ======================================================================
# Implementation Notes:
#   - Review the condition syntax and implement using your SIEM's
#     native aggregation/correlation capabilities
#   - Splunk: Use 'stats' or 'transaction' commands
#   - Elastic: Use EQL sequences or aggregation queries
#   - Sentinel: Use KQL summarize/make-series operators
# ======================================================================
