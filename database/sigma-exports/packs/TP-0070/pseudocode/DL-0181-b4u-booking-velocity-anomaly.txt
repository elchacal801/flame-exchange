# ======================================================================
# FLAME Detection Rule: B4U Booking Velocity Anomaly — Same Payment Method Booking Across Multiple OTAs/Hotels in Rapid Succession
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: high
# CFPF Phase: P4
# Threat Paths: TP-0070
#
# Description:
#   Detects Buy-for-You (B4U) booking fraud patterns where the same payment
#   method is used to make travel bookings across multiple OTAs, airlines,
#   or hotel chains within a 24-hour period, especially when bookings are
#   made for different travelers. B4U operators use stolen payment
#   credentials to purchase travel at legitimate platforms, then resell
#   bookings at 30-60% of face value. Sumsub estimates B4U services
#   generate $37B in annual travel fraud losses. The velocity of bookings
#   across disparate merchants with mismatched traveler names is the
#   hallmark detection signal.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: payments
#   service: travel_bookings
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection:
#     transaction_type:
#     - travel_booking
#     - hotel_reservation
#     - flight_purchase
#     - car_rental
#   filter_velocity:
#     booking_count_24h|gte: 3
#   filter_multi_merchant:
#     unique_merchants_24h|gte: 2
#   filter_multi_traveler:
#     unique_travelers_24h|gte: 2
#   timeframe: 24h
#   condition: 'selection AND filter_velocity AND filter_multi_merchant AND filter_multi_traveler
#   
#     '
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Corporate travel managers booking for multiple employees using a company card
#   - Travel agencies making legitimate bookings for multiple clients
#   - Family vacation bookings across multiple platforms for price comparison
#   - Event organizers booking group travel for conferences or weddings
#
# ======================================================================

#     native aggregation/correlation capabilities
#   - Splunk: Use 'stats' or 'transaction' commands
#   - Elastic: Use EQL sequences or aggregation queries
#   - Sentinel: Use KQL summarize/make-series operators
