# ======================================================================
# FLAME Detection Rule: Ghost Student Financial Aid Fraud — Multi-Signal Botnet Enrollment Detection
# ======================================================================
#
# This rule uses aggregation/correlation logic that requires
# SIEM-native implementation. It cannot be auto-converted to
# SPL/Lucene/KQL by pySigma backends.
#
# Level: high
# CFPF Phase: P4
# Threat Paths: TP-0033
#
# Description:
#   Detects ghost student financial aid fraud by correlating three
#   botnet-driven enrollment indicators: (1) multiple FAFSA applications
#   originating from the same device fingerprint or IP address within an
#   enrollment cycle, (2) shared bank accounts across distinct financial
#   aid applicants (multiple student IDs disbursing to the same
#   routing/account number), or (3) financial aid disbursements from
#   multiple student IDs flowing to the same beneficiary account or being
#   withdrawn at the same ATM cluster. California community colleges
#   flagged 31-34% of applications as potentially fraudulent, and Equifax
#   reported preventing $180M in ghost student fraud in a single year.
#   Ghost student botnets exploit open-enrollment admissions to submit
#   hundreds of synthetic identity applications per cycle, file FAFSA for
#   maximum Pell Grant eligibility ($7,395/year), and extract disbursements
#   before census date attendance verification can trigger aid return
#   requirements.
#
# ----------------------------------------------------------------------
# Log Source
# ----------------------------------------------------------------------
#   product: banking
#   service: transaction_monitoring
#
# ----------------------------------------------------------------------
# Detection Logic (Sigma YAML)
# ----------------------------------------------------------------------
#   selection_shared_device:
#     application_source: fafsa
#     applications_per_device|gte: 3
#     timeframe: 90d
#   selection_shared_bank_account:
#     disbursement_type: financial_aid
#     distinct_student_ids_per_account|gte: 2
#   selection_shared_beneficiary:
#     disbursement_type: financial_aid
#     distinct_student_ids_per_beneficiary|gte: 2
#     timeframe: 30d
#   condition: 'selection_shared_device OR selection_shared_bank_account OR selection_shared_beneficiary
#   
#     '
#
# ----------------------------------------------------------------------
# Known False Positives
# ----------------------------------------------------------------------
#   - Married couples or family members legitimately sharing a bank account for financial aid from different students
#   - Shared household devices used by multiple family members applying to the same institution
#   - Institutional computer labs or library terminals used by multiple applicants
#
